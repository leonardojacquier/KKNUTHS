# ============================================================
# Bloco do kasteller.com.py para o Caddyfile do VPS
#
# Pré-requisitos:
#   1. DNS (no painel do kasteller.com.py): registro A  @  → 187.127.13.220
#      e registro A  www → 187.127.13.220. Se o domínio um dia tiver e-mail
#      em outro servidor, NÃO tocar em MX (mesma regra da GNH).
#   2. Os arquivos já chegam sozinhos em /opt/gnh/assets/kasteller2/
#      pelo autodeploy da GNH (cron */2, copia assets/ do repo KKNUTHS).
#
# ⚠️ O Caddyfile é COMPARTILHADO. Antes de recarregar:
#      cp /etc/caddy/Caddyfile /etc/caddy/Caddyfile.bak-$(date +%F)
#      caddy validate --config /etc/caddy/Caddyfile
#      systemctl reload caddy        # reload, NUNCA restart
# ============================================================

kasteller.com.py, www.kasteller.com.py {
	root * /opt/gnh/assets/kasteller2
	encode gzip

	# HTML sempre fresco; fontes, imagens e vídeo podem cachear.
	# O curinga */ cobre as URLs de diretório (/buscador/); sem ele elas saem
	# sem Cache-Control e o navegador segura a versão velha.
	@html path / /index.html */ *.html
	header @html Cache-Control "no-cache"
	@estaticos path /fonts/* /img/* /video/* /buscador/img/* /og.jpg /favicon.*
	header @estaticos Cache-Control "public, max-age=604800"

	file_server
}
