# ============================================================
# MIGRAÇÃO — bloco do gnhorizons.com para o Caddyfile do VPS
# Torna reais as URLs canônicas: /  /ventas/  /institucional/
# SEMPRE: caddy validate ANTES de systemctl reload caddy
# Pré-requisito: DNS de gnhorizons.com (e www) apontando pro VPS
# ============================================================
gnhorizons.com, www.gnhorizons.com {
	# requer o snippet (gnh_security) no topo do Caddyfile — ver caddy-security-headers.txt
	import gnh_security
	# RAIZ = /opt/gnh, a mesma do domínio de preview. É o que os scripts de
	# deploy montam: assets/nuevo/. vai para a raiz (preserva /ventas/,
	# /fichas/, /img/, sitemap.xml já indexados) e gnh-redesign.html é copiado
	# por último para index.html, virando a home.
	# NÃO aponte para /opt/gnh/assets/nuevo: com esse root o redesign nunca é
	# servido (a home cai no site antigo) e as referências assets/nuevo/img/...
	# do redesign apontam para um caminho inexistente.
	root * /opt/gnh
	encode gzip

	# vídeos e logos do carrossel vivem fora de nuevo/ (caminhos /assets/... nas páginas)
	handle /assets/video/* {
		root * /opt/gnh
		file_server
	}
	handle /assets/img/* {
		root * /opt/gnh
		file_server
	}

	# HTML sempre fresco; estáticos com cache
	# (*/ cobre TODAS as páginas de diretório — /ventas/equipos-de-concreto/ etc.;
	#  sem isso, subpáginas ficavam sem no-cache e o navegador segurava versão velha)
	@html path / /index.html */ *.html
	header @html Cache-Control "no-cache"

	file_server
}
